class Solution:

    def findNthDigit(self, n: int) -> int:

        # 确定所求数字的数位是多少
        digit=1
        start=1
        count=9
        while n>count:
            n-=count
            digit+=1
            start*=10
            count=9*start*digit

        # 确定所求数字
        num=start+(n-1)//digit

        # 确定所在的数位
        s=str(num)
        return int(s[(n-1)%digit])
