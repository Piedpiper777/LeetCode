思路一：直接递归自己，自下而上

```python
class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int:

        if not root:
            return 0

        left_max=self.maxDepth(root.left)
        right_max=self.maxDepth(root.right)

        return max(left_max,right_max)+1
```

思路二：自上而下层序遍历，遍历一层，结果+1

```python
class Solution:

    def maxDepth(self, root: Optional[TreeNode]) -> int:

        if not root:
            return 0

        queue=[root]
        res=0

        while queue:
            tem=[] #用来记录子节点
            for node in queue:

                if node.left:
                    tem.append(node.left)
                if node.right:
                    tem.append(node.right)

            queue=tem
            res+=1

        return res
```

但是其实可以用递归的方法来遍历

```python
class Solution:

    def maxDepth(self, root: Optional[TreeNode]) -> int:

        res=0

        def dfs(node,depth):

            if not node:
                return

            nonlocal res
            depth+=1
            res=max(res,depth)

            dfs(node.left,depth)
            dfs(node.right,depth)

        dfs(root,0)
        
        return res
```
