
### 思路

把null考虑进去进行层序遍历即可

```python
class Codec:

    def serialize(self, root):

        """Encodes a tree to a single string.
        :type root: TreeNode
        :rtype: str
        """

        if not root:
            return "[]"

        dq=collections.deque()
        dq.append(root)

        res=[]

        while dq:
            tem=dq.popleft()

            if tem:
                res.append(str(tem.val))
                dq.append(tem.left)
                dq.append(tem.right)

            else:
                res.append("null")

        return '['+','.join(res)+']'

  

    def deserialize(self, data):
        """Decodes your encoded data to tree.
        :type data: str
        :rtype: TreeNode
        """

        if data =="[]":
            return

        datalist=data[1:-1].split(',')
        
        i=1

        root=TreeNode(int(datalist[0]))

        dq=collections.deque()
        dq.append(root)

        while dq:

            tem=dq.popleft()

            if datalist[i]!="null":
                tem.left=TreeNode(int(datalist[i]))
                dq.append(tem.left)

            i+=1

            if datalist[i]!="null":
                tem.right=TreeNode(int(datalist[i]))
                dq.append(tem.right)

            i+=1

        return root
```
