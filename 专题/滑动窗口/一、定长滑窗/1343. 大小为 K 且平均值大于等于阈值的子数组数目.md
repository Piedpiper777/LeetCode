```python
class Solution:

    def numOfSubarrays(self, arr: List[int], k: int, threshold: int) -> int:

        sum_wid=0
        res=0
        left=0
        n=len(arr)
        target=threshold*k

        for right in range(n):
            sum_wid+=arr[right]

            if right-left+1>k:
                sum_wid-=arr[left]
                left+=1

            if right-left+1==k:
                if sum_wid >=target:
                    res+=1

        return res
```
